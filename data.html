const AK_URL =
  'https://nikud-7eb2e-default-rtdb.europe-west1.firebasedatabase.app/cps_v1.json';

async function loadAkademia() {
  const res  = await fetch(AK_URL + '?_=' + Date.now());   // GET בלבד
  const data = await res.json();
  return (data.classes || []).map(c => ({
    className: c.name,
    students: (c.students || []).map(s => ({
      id:     s.id,
      name:   s.name,
      points: s.points      || 0,   // הארנק הנוכחי (יורד בקניות בחנות)
      total:  s.earnedTotal || s.points || 0,   // סה"כ שנצבר — לטבלת מובילים
      week:   s.weeklyPoints || 0
    })).sort((a,b) => b.total - a.total)
  }));
}

loadAkademia().then(console.log);
setInterval(() => loadAkademia().then(console.log), 10000);  // רענון כל 10 שנ'
